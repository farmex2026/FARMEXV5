var HOJA_DATOS = '_datos';
var TROZO = 45000;   // Sheets no acepta más de 50.000 caracteres por celda

/** La app pide los datos al abrirse */
function doGet() {
  var libro = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = libro.getSheetByName(HOJA_DATOS);
  var texto = '{}';

  if (hoja && hoja.getLastRow() > 0) {
    texto = hoja.getRange(1, 1, hoja.getLastRow(), 1)
                .getValues().map(function (f) { return f[0]; }).join('');
  }

  return ContentService.createTextOutput(texto)
    .setMimeType(ContentService.MimeType.JSON);
}

/** La app manda los datos cada vez que cambian */
function doPost(e) {
  var cuerpo = JSON.parse(e.postData.contents);
  var libro = SpreadsheetApp.getActiveSpreadsheet();

  // 1) el estado completo, para poder recuperarlo desde cualquier PC
  if (cuerpo.estado) {
    var hoja = libro.getSheetByName(HOJA_DATOS) || libro.insertSheet(HOJA_DATOS);
    hoja.clear();
    var texto = JSON.stringify(cuerpo.estado);
    var partes = [];
    for (var i = 0; i < texto.length; i += TROZO) {
      partes.push([texto.substr(i, TROZO)]);
    }
    if (!partes.length) partes.push(['{}']);
    hoja.getRange(1, 1, partes.length, 1).setValues(partes);
  }

  // 2) las hojas para leer con los ojos
  var d = cuerpo.planillas || {};
  volcar(libro, 'Proveedores',   d.proveedores);
  volcar(libro, 'Facturas',      d.facturas);
  volcar(libro, 'Mensuales',     d.mensuales);
  volcar(libro, 'Combustible',   d.combustible);
  volcar(libro, 'SICE',          d.sice);
  volcar(libro, 'YER',           d.yer);
  volcar(libro, 'Nota liquido',  d.nota);
  volcar(libro, 'Bancos',        d.bancos);

  var copia = libro.getSheetByName('Copia') || libro.insertSheet('Copia');
  copia.getRange('A1').setValue('Última copia');
  copia.getRange('B1').setValue(new Date());

  return ContentService
    .createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}

function volcar(libro, nombre, filas) {
  if (!filas) return;
  var hoja = libro.getSheetByName(nombre) || libro.insertSheet(nombre);
  hoja.clear();

  if (!filas.length) { hoja.getRange(1, 1).setValue('Sin datos'); return; }

  var columnas = [];
  filas.forEach(function (f) {
    Object.keys(f).forEach(function (k) {
      if (columnas.indexOf(k) === -1) columnas.push(k);
    });
  });

  var matriz = [columnas];
  filas.forEach(function (f) {
    matriz.push(columnas.map(function (c) {
      return f[c] === undefined || f[c] === null ? '' : f[c];
    }));
  });

  hoja.getRange(1, 1, matriz.length, columnas.length).setValues(matriz);
  hoja.getRange(1, 1, 1, columnas.length).setFontWeight('bold');
  hoja.setFrozenRows(1);
}
